% Compact pictogram of an audio waveform: 15 vertical bars at a
% pre-defined pseudo-envelope. Width is ~1.7cm, height ~0.45cm.
\newcommand{\WaveformPictogram}{%
  \begin{tikzpicture}[baseline=-0.6ex]
    \foreach \i/\h in {%
        0/0.10, 1/0.22, 2/0.16, 3/0.34, 4/0.24,
        5/0.42, 6/0.30, 7/0.48, 8/0.34, 9/0.40,
        10/0.22, 11/0.32, 12/0.16, 13/0.24, 14/0.12%
    }{
      \pgfmathsetmacro{\xpos}{-0.84 + \i*0.12}
      \draw[line width=0.85pt, line cap=round, draw=cEncB!90]
        (\xpos, -\h) -- (\xpos, \h);
    }
  \end{tikzpicture}%
}

\tikzset{

  audio/.style     ={module, fill=cAudio, draw=black!35},

                     minimum width=11.4cm, minimum height=0.95cm,
                     font=\sffamily},

                     font=\sffamily\small},

  cellBound/.style ={tokcell, fill=cBound, draw=cBoundB},
