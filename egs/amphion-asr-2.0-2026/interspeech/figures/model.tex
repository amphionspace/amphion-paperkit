% Architecture of OFA-TS in TS-ASR mode, drawn bottom-up in the layout of a module diagram
% (inputs at the bottom, token rows between modules, output tokens on top, legend, frozen /
% trainable marks on modules only; token colours are deeper or lighter tints of palette hues).
% Facts: ../sections/03_model.tex (shared convolution applied
% to enrollment and mixture separately, learnable separator between them, 24-layer AuT
% Transformer with one attention window, frozen convolutions and text embeddings, presence
% decision on the output language tag).
% Build: cd figures && tectonic -Z search-path=../../../../figures model.tex
% Labels are 6.5--8 pt at the final 80 mm column width.

% Waveform pictogram: \wavepic{colour}{envelope}
\newcommand{\wavepic}[2]{%
  \begin{tikzpicture}[x=1mm,y=1mm,baseline=-0.6ex]
    \foreach \h [count=\i] in #2
      \draw[line width=0.5pt,line cap=round,draw=#1] ({0.75*\i},-\h) -- ({0.75*\i},\h);
  \end{tikzpicture}}
