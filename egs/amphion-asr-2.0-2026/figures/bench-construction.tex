% fig:bench — how AmphionTS-Bench samples are built.
% Facts: refs/docs/ts_benchmark.md §1 (21,720 synthetic / 2,381 real samples),
% §2.3 (base scenes of target + 1--2 interferers; one factor changed at a time:
% enrollment, noise floor, room response, level, background; target-absent counterparts keep the
% mixture and swap the enrollment for a random person or one of the 5 speakers
% closest to an interferer), §2.4 (real corpora, no mixing, 8--22 s window from a
% sentence start extended to whole sentences up to 26 s; present = speaks >= 1 s in
% the window, absent = same recording, silent in the window; enrollment outside the
% window with a 1 s margin), §2.5 (10 s stored, first 5 s used, 0.3--0.6 s pauses,
% zero padding, 2/3/4 s evaluated separately), §4 (metrics).

% \seg{x1}{x2}{y}{style}: one sentence on a speaker track
\newcommand{\seg}[4]{\path[#4, rounded corners=1pt] (#1,#3-0.09) rectangle (#2,#3+0.09);}

% \pair{name}{y-present}{y-absent}{present label}{absent label}
\def\xE{10.7}
\newcommand{\pair}[5]{%
  \node[enr, fill=cEncB]    (#1p) at (\xE,#2) {\faUser};
  \node[enr, fill=cPromptB] (#1a) at (\xE,#3) {\faUser};
  \foreach \r/\y in {p/#2, a/#3} {
    \node[mix] (#1m\r) at (\xE+1.15,\y) {};
    \path[spkT] ($(#1m\r.west)+(0.12,0.04)$) rectangle +(0.62,0.08);
    \path[spkI] ($(#1m\r.west)+(0.55,-0.12)$) rectangle +(0.72,0.08);
    \node[font=\scriptsize, text=black!50] at ($(#1\r.east)!0.5!(#1m\r.west)$) {+};
  }
  \node[note, font=\tiny] at ($(#1mp.south)!0.5!(#1ma.north)$) {same audio};
  \node[tlab, align=right] at ($(#1p.west)+(-0.06,0)$) {#4};
  \node[tlab, align=right] at ($(#1a.west)+(-0.06,0)$) {#5};
  \node[res] at ($(#1mp.east)+(0.12,0)$) {\textcolor{cEncB}{\textbf{target present}}\\CER, empty-output rate};
  \node[res] at ($(#1ma.east)+(0.12,0)$) {\textcolor{cPromptB}{\textbf{target absent}}\\false-alarm rate};
}
