\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/capabilities-v2.png}
  \caption{AmphionASR exposes three production-oriented capabilities
    that arise from coupling a Whisper-style audio encoder with a
    pre-trained LLM. \emph{Top-left} hotword conditioning: a
    user-supplied keyword list (e.g.\ \texttt{Hotword: Amphion}) is
    placed in the LLM prompt and the decoder biases towards those
    terms. \emph{Top-right} target-speaker recognition: a
    $3$--$5$\,s enrollment utterance steers the model to transcribe
    only the matching speaker in overlapped multi-talker audio.
    \emph{Bottom} noise--RIR robustness: an augmentation curriculum
    plus a noise-robustness SFT stage keeps the transcript clean
    under additive noise and room reverberation. Quantitative results
    for each capability are reported in
    Sections~\ref{sec:experiments}--\ref{sec:noise}.}
  \label{fig:teaser}
\end{figure}
