\begin{figure}[H]
  \centering
  \includegraphics[width=\linewidth]{figures/capabilities.pdf}
  \caption{AmphionASR exposes five production-oriented Mandarin--English
    ASR capabilities through a single LLM prompt. The first row covers
    the baseline multilingual setting, mixed-language code-switching and
    prompt-driven hotword biasing; the second row covers target-speaker
    recognition (selected with a 3--5\,s enrollment utterance) and
    additive-noise / room-impulse-response / competing-speaker
    robustness. Numbers shown are the released checkpoint's reported
    results -- AISHELL-1 ($1.85$\,\% CER) and LibriSpeech
    \texttt{test-clean} ($1.82$\,\% WER) for plain ASR, TALCS
    ($7.48$\,\% MER) for code-switching, CommonVoice~17 Mandarin
    ($6.01 \to 2.67$\,\% CER with $k\!=\!5$ oracle hotwords) for
    hotword biasing, the internal hot+TS test set
    ($13.06$\,\% MER) for target-speaker recognition, and the
    CommonVoice~17 Mandarin MUSAN-speech split
    ($21.45 \to 9.45$\,\% CER after the noise-robustness SFT stage)
    for noise robustness.}
  \label{fig:teaser}
\end{figure}

% --- Teaser 2: architecture diagram -----------------------------------------
\begin{figure}[H]
  \centering
  \includegraphics[width=\linewidth]{figures/architecture.pdf}
  \caption{AmphionASR architecture. The audio branch (left) maps a
    16\,kHz waveform through a Whisper-style log-mel front-end, a
    convolutional sub-sampler, a 24-layer Whisper-style Transformer
    encoder and a lightweight modality projector into the LLM's
    2560-dimensional hidden space. The text branch (right) tokenises
    the chat template and replaces four special boundary tokens with
    learnable prompt vectors. The two streams meet at a merge step
    that substitutes the \texttt{<speech>} placeholder with the
    audio embedding sequence, after which Qwen3-4B-Instruct performs
    next-token prediction on the assistant span only. See
    Section~\ref{sec:model} for component-level details.}
  \label{fig:architecture}
\end{figure}
