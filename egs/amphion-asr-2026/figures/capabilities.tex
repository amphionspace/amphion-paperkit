% Column widths for the four-column row layout.  Tuned so that the

\tikzset{
  caprow/.style    ={draw=cBoundB!65, rounded corners=3pt,
                     line width=0.6pt, fill=cAudio!50,
                     inner sep=6pt, inner ysep=8pt,
                     text width=15.0cm,
                     minimum height=1.05cm},
}

% Body of one capability row -- four side-by-side minipages so the
% icon / name / description / prompt-key columns line up across
% every row.  Vertical centering keeps a 2-line description from
% pushing the icon out of alignment.
\newcommand{\caprow}[4]{%
  \begin{minipage}[c]{\colIcon}\centering%
    {\Huge\color{cLLMB!85!black}#1}%
  \end{minipage}%
  \hspace{0.05cm}%
  \begin{minipage}[c]{\colName}\raggedright%
    {\bfseries\small\color{cLLMB!85!black}#2}%
  \end{minipage}%
  \hspace{0.05cm}%
  \begin{minipage}[c]{\colDesc}\raggedright%
    {\footnotesize\color{black!72}#3}%
  \end{minipage}%
  \hspace{0.05cm}%
  \begin{minipage}[c]{\colPrompt}\raggedright%
    {\scriptsize\ttfamily\color{cAdpB!90!black}\textgreater\textgreater~#4}%
  \end{minipage}%
}
