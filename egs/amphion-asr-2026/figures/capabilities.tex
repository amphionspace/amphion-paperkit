% Clean waveform: 15 vertical bars at a fixed pseudo envelope.
\newcommand{\WaveformPictogram}{%
  \begin{tikzpicture}[baseline=-0.6ex]
    \foreach \i/\h in {%
        0/0.10, 1/0.22, 2/0.16, 3/0.34, 4/0.24,
        5/0.42, 6/0.30, 7/0.48, 8/0.34, 9/0.40,
        10/0.22, 11/0.32, 12/0.16, 13/0.24, 14/0.12%
    }{
      \pgfmathsetmacro{\xpos}{-0.84 + \i*0.12}
      \draw[line width=0.85pt, line cap=round, draw=cEncB!90]
        (\xpos, -\h) -- (\xpos, \h);
    }
  \end{tikzpicture}%
}

% Two-channel pictogram: short enrollment (peach, top) + mixture (teal, bottom).
\newcommand{\DualWaveformPictogram}{%
  \begin{tikzpicture}[baseline=-0.6ex]
    \node[font=\scriptsize, text=cAdpB] at (-1.30, 0.25) {enroll};
    \foreach \i/\h in {0/0.06, 1/0.14, 2/0.10, 3/0.16, 4/0.08, 5/0.12, 6/0.18, 7/0.10}{
      \pgfmathsetmacro{\xpos}{-0.55 + \i*0.10}
      \draw[line width=0.75pt, line cap=round, draw=cAdpB!85]
        (\xpos, 0.25-\h) -- (\xpos, 0.25+\h);
    }
    \node[font=\scriptsize, text=cEncB] at (-1.30, -0.30) {mix};
    \foreach \i/\h in {%
        0/0.08, 1/0.20, 2/0.14, 3/0.28, 4/0.18,
        5/0.32, 6/0.24, 7/0.36, 8/0.26, 9/0.30,
        10/0.16, 11/0.24, 12/0.14, 13/0.20, 14/0.10%
    }{
      \pgfmathsetmacro{\xpos}{-0.55 + \i*0.10}
      \draw[line width=0.80pt, line cap=round, draw=cEncB!90]
        (\xpos, -0.30-\h*0.7) -- (\xpos, -0.30+\h*0.7);
    }
  \end{tikzpicture}%
}
