% Body of a capability card -- packed inside a minipage so the inner
% \\ line breaks do not collide with TikZ's node-level row parser.
\newcommand{\capbody}[4]{%
  \begin{minipage}[c]{4.3cm}\centering%
    \bigicon{#1}\\[6pt]%
    \captitle{#2}\\[4pt]%
    \capdesc{#3}\\[7pt]%
    \promptline{#4}%
  \end{minipage}%
}
